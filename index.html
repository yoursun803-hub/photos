    <script>

        // 1. 請在此處放入你的照片網址或相對路徑檔名

        const photos = [

            'images/photo1.jpg',

            'images/photo2.jpg',

            'images/photo3.jpg',

            'images/photo4.jpg'

        ];


        // 2. 隨機顯示的溫馨文字

        const messages = [

            '要想我喔！',

            '我想妳了！',

            '愛妳'

        ];


        function startDraw() {

            // 切換畫面

            document.getElementById('start-screen').style.display = 'none';

            document.getElementById('draw-screen').style.display = 'flex';


            const card = document.getElementById('card');

            const resultText = document.getElementById('result-text');

            const redrawBtn = document.getElementById('redraw-btn');


            // 播放洗牌/震動動畫 (持續 1.5 秒)

            card.classList.add('shuffling');


            setTimeout(() => {

                // 停止動畫

                card.classList.remove('shuffling');


                // 隨機選擇照片與文字

                const randomPhoto = photos[Math.floor(Math.random() * photos.length)];

                const randomMessage = messages[Math.floor(Math.random() * messages.length)];


                // 設置圖片與文字

                document.getElementById('photo').src = randomPhoto;

                resultText.innerText = randomMessage;


                // 翻牌動畫

                card.classList.add('flipped');


                // 翻牌後顯示文字與重新抽卡按鈕

                setTimeout(() => {

                    resultText.classList.add('show');

                    redrawBtn.style.display = 'inline-block';

                }, 400);


            }, 1500); // 1.5 秒抽卡動畫時間

        }


        function resetCard() {

            const card = document.getElementById('card');

            const resultText = document.getElementById('result-text');

            const redrawBtn = document.getElementById('redraw-btn');


            // 蓋牌並隱藏文字

            card.classList.remove('flipped');

            resultText.classList.remove('show');

            redrawBtn.style.display = 'none';


            // 稍微延遲後重新開始抽卡

            setTimeout(() => {

                startDraw();

            }, 500);

        }

    </script>
